    agents.extend(page.result.data)

  return agents

FINISHED = {"done", "error", "missing_content", "noop", "self_managed"}


def read_doc(name):
  """Return the content of an embedded document, as bytes."""
  return DOCS[name].encode("utf-8")


def wait_until_processed(library_id, timeout=300):
  """Wait until Mistral has processed every document of a Library."""
  start = time.time()
  while True:
    docs = client.beta.libraries.documents.list(library_id=library_id).data
    pending = [d.name for d in docs if str(d.process_status) not in FINISHED]
    if not pending or time.time() - start > timeout:
      break
    time.sleep(5)
  for d in docs:
    print(f"  {d.name}: {d.process_status}")


def upload(library_id, doc_name, attempts=10):
  """Upload one document, waiting if the processing rate limit is reached."""
  for attempt in range(attempts):
    try:
      return client.beta.libraries.documents.upload(
        library_id=library_id,
        file=File(file_name=doc_name, content=read_doc(doc_name)),
      )
    except SDKError as error:
      if error.status_code != 429 or attempt == attempts - 1:
        raise
      print(f"  Processing rate limit reached; retrying {doc_name} in 30 s...")
      time.sleep(30)


def create_library(name, doc_names, description=""):
  """Create a Library and upload documents to it."""
  library = client.beta.libraries.create(name=name, description=description)
  for doc_name in doc_names:
    upload(library.id, doc_name)
  print(f"Created library {name} ({library.id}):")
  wait_until_processed(library.id)
  return library

## 4. Attach the Library to the Agent

An Agent uses a Library through its `document_library` **tool**. If Studio
lets you attach a Library in the Agent editor, do it there and skip this
section.

The helper below keeps the Agent's other tools (web search, code
interpreter...) and adds the selected Libraries. Mistral saves the change as
a new **version** of the Agent, so you can always go back.
=== CELL 13 code ===
def tool_as_dict(tool):
  """A tool as a plain dictionary, without empty fields."""
  if not isinstance(tool, dict):
    tool = tool.model_dump(by_alias=True, exclude_none=True)
  return {key: value for key, value in tool.items() if value is not None}


def attach_libraries(agent_id, library_ids):
  """Add Libraries to an Agent's document_library tool, keeping other tools."""
  agent = client.beta.agents.get(agent_id=agent_id)
  tools = [tool_as_dict(t) for t in (agent.tools or [])]
  other_tools = [t for t in tools if t["type"] != "document_library"]
  library_tools = [t for t in tools if t["type"] == "document_library"]
  library_tool = (
    library_tools[0] if library_tools else {"type": "document_library"}
  )
  ids = [i for t in library_tools for i in t.get("library_ids", [])]
  ids += [i for i in library_ids if i not in ids]
  library_tool = {**library_tool, "library_ids": ids}
  updated = client.beta.agents.update(
    agent_id=agent_id, tools=other_tools + [library_tool]
  )
  print("Tools before:", [t["type"] for t in tools])
  print("Tools after: ", [tool_as_dict(t)["type"] for t in updated.tools])
  print(f"Libraries attached: {ids}")
  print(f"Agent version {agent.version} -> {updated.version}")
  return updated


agent = attach_libraries(agent.id, library_ids)

## 5. Ask the Agent a question

Use a **concrete** question whose answer you know is in the documents. Vague
requests ("tell me something from my library") test browsing, not retrieval.
The output shows the tools the Agent used, its answer, and its sources. The
sources are not always returned, even when the Library was searched: check
"Tools used", and the source named in the answer.
=== CELL 16 code ===
def ask(agent_id, question, show=True):
  """Ask an Agent one question in a new conversation."""
  response = client.beta.conversations.start(agent_id=agent_id, inputs=question)
  tools_used, answer, sources = [], [], []
  for entry in response.outputs:
    if entry.type == "tool.execution":
      tools_used.append(f"{entry.name} {entry.arguments}")
    elif entry.type == "message.output":
      content = entry.content
      if isinstance(content, str):
        answer.append(content)
        continue
      for chunk in content:
        if chunk.type == "text":
          answer.append(chunk.text)
        elif chunk.type == "tool_reference":
          sources.append(chunk.title)
  result = {
    "answer": "".join(answer).strip(),
    "tools": tools_used,
    "sources": sorted(set(sources)),
    "response": response,
  }
  if show:
    print("Tools used:", *result["tools"] or ["none"], sep="\n  ")
    print("\nAnswer:\n" + result["answer"])
    print("\nSources:", ", ".join(result["sources"]) or "none returned")
  return result

def make_test_agent(name, library_ids, base_agent):
  """Create a temporary Agent like base_agent, using the given Libraries."""
  return client.beta.agents.create(
    model=base_agent.model,
    name=name,
    instructions=base_agent.instructions,
    tools=[{"type": "document_library", "library_ids": library_ids}],
  )

## 11. Going further

Libraries are a **managed** RAG system: Mistral manages most low-level
retrieval choices (extraction, chunking, embeddings, ranking). The API
exposes a few controls, while the lower-level search tooling exposes
substantially more:

* [Libraries documentation](https://docs.mistral.ai/studio/search/libraries)
  and [Agents & Conversations](https://docs.mistral.ai/studio/agents/agents-api);
* [Agent tools](https://docs.mistral.ai/studio/agents/agent-tools), including
  your own functions;
* [Search Toolkit](https://docs.mistral.ai/studio/search/search-toolkit), for
  building your own retrieval pipeline.
